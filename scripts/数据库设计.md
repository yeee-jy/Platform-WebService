### 二、Python SQLAlchemy ORM 模型
```python
from datetime import datetime
from sqlalchemy import (
    Column, String, Enum, Date, TIMESTAMP, ForeignKey, Table, Integer,
    create_engine, UniqueConstraint
)
from sqlalchemy.orm import relationship, declarative_base, sessionmaker

Base = declarative_base()

# 关联表定义
user_role = Table(
    'user_role', Base.metadata,
    Column('user_id', String(20), ForeignKey('users.user_id', ondelete='CASCADE', onupdate='CASCADE'), primary_key=True),
    Column('role_id', Integer, ForeignKey('roles.role_id', ondelete='CASCADE', onupdate='CASCADE'), primary_key=True)
)

role_permission = Table(
    'role_permission', Base.metadata,
    Column('role_id', Integer, ForeignKey('roles.role_id', ondelete='CASCADE', onupdate='CASCADE'), primary_key=True),
    Column('perm_id', Integer, ForeignKey('permissions.perm_id', ondelete='CASCADE', onupdate='CASCADE'), primary_key=True)
)

class User(Base):
    __tablename__ = 'users'
    
    user_id = Column(String(20), primary_key=True)
    password_hash = Column(String(128), nullable=False)
    name = Column(String(50), nullable=False)
    gender = Column(Enum('M', 'F', 'U', name='gender'), default='U', nullable=False)
    birthdate = Column(Date)
    college = Column(String(50))
    grade = Column(String(10))
    status = Column(Enum('ENABLED', 'DISABLED', name='user_status'), default='ENABLED', nullable=False)
    last_login_at = Column(TIMESTAMP)
    last_login_ip = Column(String(45))
    created_at = Column(TIMESTAMP, default=datetime.now, nullable=False)
    created_by = Column(String(20), nullable=False)
    updated_at = Column(TIMESTAMP, default=datetime.now, onupdate=datetime.now, nullable=False)
    updated_by = Column(String(20), nullable=False)
    
    roles = relationship('Role', secondary=user_role, back_populates='users')

class Role(Base):
    __tablename__ = 'roles'
    
    role_id = Column(Integer, primary_key=True, autoincrement=True)
    role_name = Column(String(20), unique=True, nullable=False)
    description = Column(String(255))
    
    users = relationship('User', secondary=user_role, back_populates='roles')
    permissions = relationship('Permission', secondary=role_permission, back_populates='roles')

class Permission(Base):
    __tablename__ = 'permissions'
    
    perm_id = Column(Integer, primary_key=True, autoincrement=True)
    perm_name = Column(String(50), unique=True, nullable=False)
    description = Column(String(255))
    
    roles = relationship('Role', secondary=role_permission, back_populates='permissions')

# 使用示例
engine = create_engine('mysql+pymysql://user:password@host:port/dbname')
Base.metadata.create_all(engine)
Session = sessionmaker(bind=engine)
```

### 三、设计说明
1. **RBAC 模型**：通过角色-权限关联实现灵活的权限管理，支持动态调整权限分配
2. **性能优化**：
   - 使用`utf8mb4`字符集支持完整 Unicode
   - 为常用查询字段（college、status）添加索引
   - 使用`ON DELETE CASCADE`保持数据一致性
3. **安全设计**：
   - 密码字段长度支持 SHA512 哈希（128字符）
   - 最后登录IP支持IPv6地址存储（45字符）
4. **审计字段**：
   - 自动维护 created_at/updated_at 时间戳
   - 记录操作者 created_by/updated_by
5. **扩展性**：
   - 角色和权限可动态配置
   - 支持多角色分配（通过 user_role 表）

该方案在保证规范化的同时考虑了实际业务需求，能够满足教育系统用户管理的典型场景。
